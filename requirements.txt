ruamel.yaml       # salva o config.yaml sem apagar os comentarios
pystray           # icone do Mestre perto do relogio do Windows
pillow            # imagem do icone
pycaw             # volume so do Spotify (mixer do Windows)

playwright        # YouTube controlado por voz (usa o Edge do PC; nao baixa navegador)
kokoro-onnx       # voz Kokoro: natural e rapida, roda no PC (baixa o modelo de 330 MB uma vez)
torch             # base do reconhecimento da SUA voz (versao so processador, ~540 MB instalado)

speechbrain       # "responder so a minha voz": compara a voz com o cadastro (modelo de 85 MB, gratis, local)
